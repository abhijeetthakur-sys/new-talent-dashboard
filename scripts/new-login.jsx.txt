  // ── ui-kit login constants (fixed, not theme-aware — per kit spec) ──
  const L_PINK = "#E91E8C";
  const L_DARK = "#0D0A07";
  const L_PAPER = "#F1ECE2";
  const L_INK = "#15120D";
  const L_SUB = "#6F6657";
  const L_HAIR = "#DED6C7";
  const ROLE_BADGE = {
    admin:  { label: "Admin",  color: "#E91E8C" },
    editor: { label: "Editor", color: "#1F6F5C" },
    jd:     { label: "JD",     color: "#D4501E" },
    ops:    { label: "Ops",    color: "#6B3FA0" },
    viewer: { label: "Viewer", color: "#6F6657" },
  };
  const [focused, setFocused] = useState(null);

  function attemptWithShake() {
    const key = username.trim().toLowerCase();
    const u = USERS[key];
    if (!u || u.password !== password) {
      setErr("Incorrect username or password.");
      const form = document.getElementById("login-form");
      if (form) { form.style.animation = "none"; requestAnimationFrame(() => { form.style.animation = "loginShake 0.38s cubic-bezier(.36,.07,.19,.97)"; }); }
      return;
    }
    attempt();
  }

  const autofill = (key) => {
    const u = USERS[key];
    if (!u) return;
    setUsername(key);
    setPassword(u.password);
    setErr("");
  };

  const inputStyle = (name) => ({
    width: "100%", boxSizing: "border-box",
    background: L_PAPER,
    border: `1.5px solid ${focused === name ? L_PINK : L_HAIR}`,
    borderRadius: 8,
    padding: "13px 14px",
    fontFamily: "'Archivo', sans-serif",
    fontSize: 14,
    color: L_INK,
    outline: "none",
    transition: "border-color 0.2s ease, box-shadow 0.2s ease",
    boxShadow: focused === name ? `0 0 0 3px ${L_PINK}18` : "none",
  });

  return (
    <div className="login-grid" style={{ minHeight: "100vh", display: "grid", gridTemplateColumns: "55% 45%", background: L_DARK, fontFamily: "'Archivo',sans-serif" }}>
      <style>{`
        @import url('https://fonts.googleapis.com/css2?family=Bricolage+Grotesque:opsz,wght@12..96,400;12..96,500;12..96,700;12..96,800&family=Archivo:wght@400;500;600;700;800&family=IBM+Plex+Mono:wght@400;500;600&display=swap');
        @keyframes eq1 { 0%,100%{height:10px} 30%{height:44px} 65%{height:22px} }
        @keyframes eq2 { 0%,100%{height:28px} 20%{height:8px}  55%{height:52px} }
        @keyframes eq3 { 0%,100%{height:38px} 40%{height:14px} 72%{height:48px} }
        @keyframes eq4 { 0%,100%{height:16px} 25%{height:48px} 60%{height:28px} }
        @keyframes eq5 { 0%,100%{height:24px} 50%{height:10px} 80%{height:42px} }
        @keyframes eq6 { 0%,100%{height:34px} 15%{height:52px} 70%{height:18px} }
        @keyframes eq7 { 0%,100%{height:12px} 45%{height:38px} 85%{height:24px} }
        @keyframes spin { to { transform: rotate(360deg); } }
        @keyframes fadeSlideUp { from{opacity:0;transform:translateY(8px)} to{opacity:1;transform:translateY(0)} }
        @keyframes loginPanelL { from{opacity:0;transform:translateX(-18px)} to{opacity:1;transform:translateX(0)} }
        @keyframes loginPanelR { from{opacity:0;transform:translateX(18px)} to{opacity:1;transform:translateX(0)} }
        @keyframes loginLogo { from{opacity:0;transform:translateY(14px)} to{opacity:1;transform:translateY(0)} }
        @keyframes loginPill { from{opacity:0;transform:translateY(8px) scale(.96)} to{opacity:1;transform:none} }
        @keyframes loginShake {
          0%,100% { transform:translateX(0); }
          20% { transform:translateX(-8px); }
          40% { transform:translateX(8px); }
          60% { transform:translateX(-5px); }
          80% { transform:translateX(5px); }
        }
        @media(max-width:768px){ .login-left{display:none!important} .login-grid{grid-template-columns:1fr!important} }
      `}</style>

      {/* ── LEFT: brand panel ── */}
      <div className="login-left" style={{ position: "relative", display: "flex", flexDirection: "column", justifyContent: "space-between", padding: "44px 52px", overflow: "hidden", animation: "loginPanelL 0.7s cubic-bezier(.2,.75,.25,1) both" }}>

        {/* Background texture */}
        <div style={{ position: "absolute", inset: 0, backgroundImage: "radial-gradient(circle at 20% 80%, #E91E8C08 0%, transparent 60%), radial-gradient(circle at 80% 20%, #E0531F06 0%, transparent 50%)", pointerEvents: "none" }} />
        {/* Grid lines */}
        <div style={{ position: "absolute", inset: 0, backgroundImage: "linear-gradient(rgba(255,255,255,0.025) 1px, transparent 1px), linear-gradient(90deg, rgba(255,255,255,0.025) 1px, transparent 1px)", backgroundSize: "40px 40px", pointerEvents: "none" }} />

        {/* Logo */}
        <div style={{ animation: "loginLogo 0.8s 0.1s cubic-bezier(.2,.75,.25,1) both", position: "relative", zIndex: 1 }}>
          <img src={LOGO_ARTIUM_FULL} alt="Artium" style={{ height: 52, width: "auto", objectFit: "contain", objectPosition: "left center", mixBlendMode: "screen" }} />
        </div>

        {/* Center content */}
        <div style={{ animation: "loginLogo 0.8s 0.2s cubic-bezier(.2,.75,.25,1) both", position: "relative", zIndex: 1 }}>
          <div style={{ display: "flex", alignItems: "center", gap: 16, marginBottom: 28 }}>
            {/* Animated equalizer — 7 bars */}
            <div style={{ display: "flex", alignItems: "flex-end", gap: 6, height: 60 }}>
              {[["10px", "eq1", "1.6s"], ["28px", "eq2", "2.1s"], ["38px", "eq3", "1.8s"], ["16px", "eq4", "1.5s"], ["24px", "eq5", "2.3s"], ["34px", "eq6", "1.7s"], ["12px", "eq7", "2.0s"]].map(([ht, anim, dur], i) => (
                <div key={i} style={{ width: 5, height: ht, borderRadius: 2.5, background: `linear-gradient(to top, ${L_PINK}, ${L_PINK}99)`, animationName: anim, animationDuration: dur, animationTimingFunction: "ease-in-out", animationIterationCount: "infinite" }} />
              ))}
            </div>
          </div>
          <div style={{ fontFamily: "'Bricolage Grotesque', sans-serif", fontSize: 52, fontWeight: 800, letterSpacing: "-0.04em", color: "#fff", lineHeight: 0.95, marginBottom: 16 }}>
            Talent<br />
            <span style={{ color: L_PINK }}>Command.</span>
          </div>
          <div style={{ fontFamily: "'IBM Plex Mono', monospace", fontSize: 11, letterSpacing: "0.18em", textTransform: "uppercase", color: "rgba(255,255,255,0.35)", lineHeight: 1.8 }}>
            Artium Academy · Internal Tool<br />
            ATDP · unMute · Originals
          </div>
        </div>

        {/* Stat strip — live data */}
        <div style={{ display: "flex", gap: 28, animation: "loginLogo 0.8s 0.35s cubic-bezier(.2,.75,.25,1) both", position: "relative", zIndex: 1, flexWrap: "wrap" }}>
          {stats.map((s) => (
            <div key={s.label} style={{ borderLeft: `2px solid ${L_PINK}50`, paddingLeft: 14 }}>
              <div style={{ fontFamily: "'Bricolage Grotesque', sans-serif", fontSize: 26, fontWeight: 800, letterSpacing: "-0.03em", color: "#fff" }}>{s.n}</div>
              <div style={{ fontFamily: "'IBM Plex Mono', monospace", fontSize: 9, letterSpacing: "0.15em", textTransform: "uppercase", color: "rgba(255,255,255,0.35)", marginTop: 2 }}>{s.label}</div>
            </div>
          ))}
          <div style={{ borderLeft: `2px solid ${L_PINK}50`, paddingLeft: 14 }}>
            <div style={{ fontFamily: "'IBM Plex Mono', monospace", fontSize: 9, letterSpacing: "0.12em", textTransform: "uppercase", color: "rgba(255,255,255,0.25)", marginTop: 2 }}>Season</div>
            <div style={{ fontFamily: "'IBM Plex Mono', monospace", fontSize: 10, letterSpacing: "0.06em", color: L_PINK, marginTop: 2 }}>2025–26</div>
          </div>
        </div>
      </div>

      {/* ── RIGHT: form panel ── */}
      <div className="login-right" style={{ background: L_PAPER, display: "flex", flexDirection: "column", justifyContent: "center", padding: "52px 52px", animation: "loginPanelR 0.7s 0.08s cubic-bezier(.2,.75,.25,1) both", position: "relative", overflowY: "auto" }}>

        {/* Small logo icon top-right corner */}
        <div style={{ position: "absolute", top: 28, right: 28 }}>
          <img src={LOGO_ARTIUM} alt="Artium" style={{ height: 32, objectFit: "cover", objectPosition: "left center", width: 32, borderRadius: 8 }} />
        </div>

        <div id="login-form" style={{ maxWidth: 360, width: "100%", margin: "0 auto" }}>
          {/* Heading */}
          <div style={{ fontFamily: "'Bricolage Grotesque', sans-serif", fontSize: 40, fontWeight: 800, letterSpacing: "-0.035em", color: L_INK, lineHeight: 0.96, marginBottom: 6 }}>
            Welcome<br />back.
          </div>
          <div style={{ fontFamily: "'Archivo', sans-serif", fontSize: 14, color: L_SUB, marginBottom: 36, lineHeight: 1.5 }}>
            Sign in to access Talent Command.
          </div>

          {/* Fields */}
          <div style={{ display: "flex", flexDirection: "column", gap: 16 }}>
            <div>
              <div style={{ fontFamily: "'IBM Plex Mono', monospace", fontSize: 9.5, letterSpacing: "0.14em", textTransform: "uppercase", color: L_SUB, marginBottom: 7 }}>Username</div>
              <input
                value={username}
                onChange={(e) => { setUsername(e.target.value); setErr(""); }}
                onKeyDown={(e) => e.key === "Enter" && attemptWithShake()}
                onFocus={() => setFocused("user")}
                onBlur={() => setFocused(null)}
                placeholder="abhijeet"
                autoComplete="username"
                autoFocus
                style={inputStyle("user")}
              />
            </div>
            <div>
              <div style={{ fontFamily: "'IBM Plex Mono', monospace", fontSize: 9.5, letterSpacing: "0.14em", textTransform: "uppercase", color: L_SUB, marginBottom: 7 }}>Password</div>
              <div style={{ position: "relative" }}>
                <input
                  type={showPwd ? "text" : "password"}
                  value={password}
                  onChange={(e) => { setPassword(e.target.value); setErr(""); }}
                  onKeyDown={(e) => e.key === "Enter" && attemptWithShake()}
                  onFocus={() => setFocused("pass")}
                  onBlur={() => setFocused(null)}
                  placeholder="••••••••••"
                  autoComplete="current-password"
                  style={{ ...inputStyle("pass"), paddingRight: 44 }}
                />
                <button onClick={() => setShowPwd((p) => !p)} tabIndex={-1} type="button"
                  style={{ position: "absolute", right: 13, top: "50%", transform: "translateY(-50%)", background: "none", border: "none", cursor: "pointer", color: L_SUB, padding: 4, display: "flex", lineHeight: 1 }}>
                  <svg width="16" height="16" viewBox="0 0 22 22" fill="none" stroke="currentColor" strokeWidth="1.7" strokeLinecap="round" strokeLinejoin="round">
                    {showPwd ?
                      <><path d="M1 11s4-7 10-7 10 7 10 7-4 7-10 7-10-7-10-7z" /><circle cx="11" cy="11" r="3" /></> :
                      <><path d="M17.9 17.9A10 10 0 0 1 11 19c-6 0-10-7-10-7a17.6 17.6 0 0 1 5.1-5.9M9.9 5c.36-.07.73-.1 1.1-.1 6 0 10 7 10 7a17.6 17.6 0 0 1-2.3 3.4" /><path d="M14.1 14.1A3 3 0 0 1 7.9 7.9" /><path d="M2 2l18 18" /></>
                    }
                  </svg>
                </button>
              </div>
            </div>
          </div>

          {/* Error */}
          {err &&
            <div style={{ marginTop: 14, padding: "10px 14px", background: "#E91E8C10", border: "1px solid #E91E8C40", borderRadius: 7, fontFamily: "'Archivo', sans-serif", fontSize: 13, color: "#C0134F", animation: "fadeSlideUp 0.25s ease both" }}>
              {err}
            </div>
          }

          {/* Submit */}
          <button
            onClick={attemptWithShake}
            disabled={loading}
            style={{ display: "flex", alignItems: "center", justifyContent: "center", gap: 10, width: "100%", marginTop: 22, padding: "14px 20px", background: loading ? "#C9A88A" : L_INK, color: L_PAPER, border: "none", borderRadius: 8, fontFamily: "'Bricolage Grotesque', sans-serif", fontSize: 16, fontWeight: 700, cursor: loading ? "not-allowed" : "pointer", transition: "background 0.2s ease, transform 0.15s ease", letterSpacing: "-0.01em" }}
            onMouseEnter={(e) => { if (!loading) e.currentTarget.style.background = L_PINK; }}
            onMouseLeave={(e) => { if (!loading) e.currentTarget.style.background = L_INK; }}>
            {loading ?
              <><span style={{ width: 16, height: 16, border: "2px solid rgba(255,255,255,.3)", borderTopColor: "#fff", borderRadius: "50%", display: "inline-block", animation: "spin 0.7s linear infinite" }} /> Signing in…</> :
              "Sign in →"
            }
          </button>

          {/* Divider */}
          <div style={{ display: "flex", alignItems: "center", gap: 12, margin: "28px 0 18px" }}>
            <div style={{ flex: 1, height: 1, background: L_HAIR }} />
            <span style={{ fontFamily: "'IBM Plex Mono', monospace", fontSize: 9, letterSpacing: "0.15em", textTransform: "uppercase", color: L_SUB }}>Quick access</span>
            <div style={{ flex: 1, height: 1, background: L_HAIR }} />
          </div>

          {/* User pills — click to autofill */}
          <div style={{ display: "flex", flexWrap: "wrap", gap: 7 }}>
            {Object.entries(USERS).map(([key, u], i) => {
              const b = ROLE_BADGE[u.role] || ROLE_BADGE.viewer;
              return (
                <button key={key} onClick={() => autofill(key)}
                  style={{ display: "flex", alignItems: "center", gap: 6, padding: "6px 11px 6px 8px", borderRadius: 99, background: "transparent", border: `1px solid ${L_HAIR}`, cursor: "pointer", fontFamily: "'Archivo', sans-serif", fontSize: 12.5, fontWeight: 600, color: L_INK, transition: "all 0.15s ease", animation: `loginPill 0.4s ${0.3 + i * 0.06}s cubic-bezier(.2,.75,.25,1) both` }}
                  onMouseEnter={(e) => { e.currentTarget.style.borderColor = b.color; e.currentTarget.style.background = b.color + "12"; }}
                  onMouseLeave={(e) => { e.currentTarget.style.borderColor = L_HAIR; e.currentTarget.style.background = "transparent"; }}>
                  <span style={{ width: 8, height: 8, borderRadius: "50%", background: b.color, flexShrink: 0 }} />
                  {u.name}
                  <span style={{ fontFamily: "'IBM Plex Mono', monospace", fontSize: 8.5, letterSpacing: "0.06em", textTransform: "uppercase", color: b.color, opacity: .8 }}>{b.label}</span>
                </button>
              );
            })}
          </div>
        </div>
      </div>
    </div>);

}
